# SCP를 사용한 특정 파일 업로드 Shell 스크립트

SCP(Secure Copy Protocol)를 사용하여 특정 파일만 업로드하는 shell 스크립트를 구현하는 방법을 상세히 설명해 드리겠습니다.

## 기본 스크립트

```bash
#!/bin/bash

# SCP 파일 업로드 스크립트
# 사용법: ./upload_script.sh [로컬파일] [원격지경로]

# 변수 설정
LOCAL_FILE="$1"
REMOTE_PATH="$2"
REMOTE_USER="username"
REMOTE_HOST="remote.server.com"
REMOTE_PORT="22"  # 기본 포트는 22
IDENTITY_FILE="$HOME/.ssh/id_rsa"  # SSH 키 파일

# 필수 매개변수 확인
if [ $# -ne 2 ]; then
    echo "사용법: $0 [로컬파일] [원격지경로]"
    echo "예: $0 /path/to/local/file.txt /remote/path/"
    exit 1
fi

# 파일 존재 여부 확인
if [ ! -f "$LOCAL_FILE" ]; then
    echo "오류: 로컬 파일 '$LOCAL_FILE'을 찾을 수 없습니다."
    exit 1
fi

# SCP 명령어 실행
scp -P "$REMOTE_PORT" -i "$IDENTITY_FILE" "$LOCAL_FILE" "${REMOTE_USER}@${REMOTE_HOST}:${REMOTE_PATH}"

# 실행 결과 확인
if [ $? -eq 0 ]; then
    echo "성공: $LOCAL_FILE → ${REMOTE_USER}@${REMOTE_HOST}:${REMOTE_PATH}"
else
    echo "오류: 파일 업로드에 실패했습니다."
    exit 1
fi
```

## 고급 기능 포함 스크립트

```bash
#!/bin/bash

# 고급 SCP 파일 업로드 스크립트
# 설정 파일을 통한 유연한 관리

# 설정 파일 로드 (존재하는 경우)
CONFIG_FILE="$HOME/.scp_upload_config"
if [ -f "$CONFIG_FILE" ]; then
    source "$CONFIG_FILE"
fi

# 기본값 설정
REMOTE_USER=${REMOTE_USER:-"username"}
REMOTE_HOST=${REMOTE_HOST:-"remote.server.com"}
REMOTE_PORT=${REMOTE_PORT:-"22"}
IDENTITY_FILE=${IDENTITY_FILE:-"$HOME/.ssh/id_rsa"}
BACKUP_DIR=${BACKUP_DIR:-"$HOME/backups"}
LOG_FILE=${LOG_FILE:-"/var/log/scp_upload.log"}

# 함수 정의
log_message() {
    echo "$(date '+%Y-%m-%d %H:%M:%S') - $1" | tee -a "$LOG_FILE"
}

backup_file() {
    local file="$1"
    local backup_path="$BACKUP_DIR/$(date '+%Y%m%d_%H%M%S')_$(basename "$file")"
    
    mkdir -p "$BACKUP_DIR"
    cp "$file" "$backup_path"
    log_message "백업 생성: $file → $backup_path"
}

validate_ssh_connection() {
    ssh -p "$REMOTE_PORT" -i "$IDENTITY_FILE" -q -o BatchMode=yes -o ConnectTimeout=5 \
        "${REMOTE_USER}@${REMOTE_HOST}" exit
    
    if [ $? -ne 0 ]; then
        log_message "오류: SSH 연결 실패 - ${REMOTE_USER}@${REMOTE_HOST}"
        return 1
    fi
    return 0
}

upload_file() {
    local local_file="$1"
    local remote_path="$2"
    
    # 파일 존재 확인
    if [ ! -f "$local_file" ]; then
        log_message "오류: 파일을 찾을 수 없음 - $local_file"
        return 1
    fi
    
    # 파일 크기 확인
    local file_size=$(stat -c%s "$local_file")
    if [ "$file_size" -gt 104857600 ]; then  # 100MB 이상
        log_message "경고: 대용량 파일 업로드 - $(($file_size/1048576))MB"
    fi
    
    # 백업 생성
    backup_file "$local_file"
    
    # SCP 업로드
    log_message "업로드 시작: $local_file → ${REMOTE_USER}@${REMOTE_HOST}:$remote_path"
    
    scp -P "$REMOTE_PORT" -i "$IDENTITY_FILE" -C "$local_file" \
        "${REMOTE_USER}@${REMOTE_HOST}:$remote_path"
    
    if [ $? -eq 0 ]; then
        log_message "성공: 파일 업로드 완료"
        return 0
    else
        log_message "오류: 파일 업로드 실패"
        return 1
    fi
}

# 메인 실행 부분
main() {
    if [ $# -ne 2 ]; then
        echo "사용법: $0 [로컬파일] [원격지경로]"
        echo "예: $0 /path/to/file.txt /remote/directory/"
        echo "환경변수나 $CONFIG_FILE 파일을 통해 설정을 변경할 수 있습니다."
        exit 1
    fi
    
    LOCAL_FILE="$1"
    REMOTE_PATH="$2"
    
    # SSH 연결 확인
    if ! validate_ssh_connection; then
        exit 1
    fi
    
    # 파일 업로드
    if upload_file "$LOCAL_FILE" "$REMOTE_PATH"; then
        echo "업로드가 성공적으로 완료되었습니다."
        exit 0
    else
        echo "업로드 중 오류가 발생했습니다."
        exit 1
    fi
}

# 스크립트 실행
main "$@"
```

## 여러 파일 업로드 스크립트

```bash
#!/bin/bash

# 여러 파일 업로드 스크립트

# 설정
REMOTE_USER="username"
REMOTE_HOST="remote.server.com"
REMOTE_BASE_DIR="/remote/path"
LOCAL_BASE_DIR="/local/path"
FILE_PATTERNS=("*.txt" "*.log" "*.csv")  # 업로드할 파일 패턴

# 파일 찾기 및 업로드
upload_files() {
    for pattern in "${FILE_PATTERNS[@]}"; do
        while IFS= read -r -d '' file; do
            # 원격 경로 생성 (로컬 경로 구조 유지)
            relative_path="${file#$LOCAL_BASE_DIR/}"
            remote_path="$REMOTE_BASE_DIR/$(dirname "$relative_path")"
            
            echo "업로드: $file → ${REMOTE_USER}@${REMOTE_HOST}:$remote_path/"
            
            # 디렉토리 생성 (필요한 경우)
            ssh "${REMOTE_USER}@${REMOTE_HOST}" "mkdir -p '$remote_path'"
            
            # 파일 업로드
            scp "$file" "${REMOTE_USER}@${REMOTE_HOST}:$remote_path/"
            
            if [ $? -eq 0 ]; then
                echo "✓ 성공: $(basename "$file")"
                # 업로드 성공 후 로컬 파일 삭제 (선택사항)
                # rm "$file"
            else
                echo "✗ 실패: $(basename "$file")"
            fi
            
        done < <(find "$LOCAL_BASE_DIR" -name "$pattern" -type f -print0)
    done
}

# 메인 실행
echo "파일 업로드를 시작합니다..."
upload_files
echo "업로드 완료."
```

## 사용법

1. **스크립트 저장**: 원하는 이름으로 파일 저장 (예: `scp_upload.sh`)
2. **실행 권한 부여**: `chmod +x scp_upload.sh`
3. **실행**: `./scp_upload.sh [로컬파일] [원격지경로]`

## 주의사항

- SSH 키 기반 인증 설정 필요
- 원격 서버의 디렉토리 권한 확인
- 대용량 파일 전송 시 `-C` 옵션(압축) 사용 고려
- 방화벽 설정 확인

이 스크립트들은 기본적인 에러 처리, 로깅, 백업 기능을 포함하고 있어 프로덕션 환경에서도 사용할 수 있습니다.
