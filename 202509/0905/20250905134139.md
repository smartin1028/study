### 기본 서블릿 클래스
```java
@WebServlet("/api/*")
public class RestApiServlet extends HttpServlet {
    
    private ObjectMapper objectMapper = new ObjectMapper();
    
    @Override
    protected void doGet(HttpServletRequest req, HttpServletResponse resp) 
            throws ServletException, IOException {
        
        String pathInfo = req.getPathInfo();
        resp.setContentType("application/json");
        resp.setCharacterEncoding("UTF-8");
        
        if (pathInfo == null || pathInfo.equals("/")) {
            // 모든 사용자 조회
            List<User> users = getUserService().getAllUsers();
            objectMapper.writeValue(resp.getWriter(), users);
        } else {
            // 특정 사용자 조회
            String[] splits = pathInfo.split("/");
            if (splits.length != 2) {
                resp.sendError(HttpServletResponse.SC_BAD_REQUEST);
                return;
            }
            
            try {
                Long id = Long.parseLong(splits[1]);
                User user = getUserService().getUserById(id);
                if (user != null) {
                    objectMapper.writeValue(resp.getWriter(), user);
                } else {
                    resp.sendError(HttpServletResponse.SC_NOT_FOUND);
                }
            } catch (NumberFormatException e) {
                resp.sendError(HttpServletResponse.SC_BAD_REQUEST);
            }
        }
    }
    
    @Override
    protected void doPost(HttpServletRequest req, HttpServletResponse resp) 
            throws ServletException, IOException {
        
        try {
            User user = objectMapper.readValue(req.getReader(), User.class);
            User createdUser = getUserService().createUser(user);
            
            resp.setStatus(HttpServletResponse.SC_CREATED);
            resp.setContentType("application/json");
            objectMapper.writeValue(resp.getWriter(), createdUser);
        } catch (Exception e) {
            resp.sendError(HttpServletResponse.SC_BAD_REQUEST);
        }
    }
    
    @Override
    protected void doPut(HttpServletRequest req, HttpServletResponse resp) 
            throws ServletException, IOException {
        
        String pathInfo = req.getPathInfo();
        if (pathInfo == null || pathInfo.split("/").length != 2) {
            resp.sendError(HttpServletResponse.SC_BAD_REQUEST);
            return;
        }
        
        try {
            Long id = Long.parseLong(pathInfo.split("/")[1]);
            User user = objectMapper.readValue(req.getReader(), User.class);
            user.setId(id);
            
            User updatedUser = getUserService().updateUser(user);
            if (updatedUser != null) {
                objectMapper.writeValue(resp.getWriter(), updatedUser);
            } else {
                resp.sendError(HttpServletResponse.SC_NOT_FOUND);
            }
        } catch (NumberFormatException e) {
            resp.sendError(HttpServletResponse.SC_BAD_REQUEST);
        }
    }
    
    @Override
    protected void doDelete(HttpServletRequest req, HttpServletResponse resp) 
            throws ServletException, IOException {
        
        String pathInfo = req.getPathInfo();
        if (pathInfo == null || pathInfo.split("/").length != 2) {
            resp.sendError(HttpServletResponse.SC_BAD_REQUEST);
            return;
        }
        
        try {
            Long id = Long.parseLong(pathInfo.split("/")[1]);
            boolean deleted = getUserService().deleteUser(id);
            if (deleted) {
                resp.setStatus(HttpServletResponse.SC_NO_CONTENT);
            } else {
                resp.sendError(HttpServletResponse.SC_NOT_FOUND);
            }
        } catch (NumberFormatException e) {
            resp.sendError(HttpServletResponse.SC_BAD_REQUEST);
        }
    }
    
    private UserService getUserService() {
        // UserService 인스턴스 반환
        return new UserService();
    }
}
```

### 서비스 클래스
```java
public class UserService {
    private Map<Long, User> users = new ConcurrentHashMap<>();
    private AtomicLong idCounter = new AtomicLong(1);
    
    public List<User> getAllUsers() {
        return new ArrayList<>(users.values());
    }
    
    public User getUserById(Long id) {
        return users.get(id);
    }
    
    public User createUser(User user) {
        Long id = idCounter.getAndIncrement();
        user.setId(id);
        users.put(id, user);
        return user;
    }
    
    public User updateUser(User user) {
        if (users.containsKey(user.getId())) {
            users.put(user.getId(), user);
            return user;
        }
        return null;
    }
    
    public boolean deleteUser(Long id) {
        return users.remove(id) != null;
    }
}
```

## 2. Basic Auth 추가 방법

### Basic Auth 필터 클래스
```java
@WebFilter("/api/*")
public class BasicAuthFilter implements Filter {
    
    // 간단한 사용자 저장소 (실제로는 데이터베이스 사용 권장)
    private static final Map<String, String> USERS = new HashMap<>();
    static {
        USERS.put("admin", "password123"); // username: admin, password: password123
        USERS.put("user", "userpass");     // username: user, password: userpass
    }
    
    @Override
    public void doFilter(ServletRequest request, ServletResponse response, 
                         FilterChain chain) throws IOException, ServletException {
        
        HttpServletRequest req = (HttpServletRequest) request;
        HttpServletResponse resp = (HttpServletResponse) response;
        
        String authHeader = req.getHeader("Authorization");
        
        if (authHeader != null && authHeader.startsWith("Basic ")) {
            String base64Credentials = authHeader.substring("Basic ".length());
            String credentials = new String(Base64.getDecoder().decode(base64Credentials), 
                                          StandardCharsets.UTF_8);
            
            final String[] values = credentials.split(":", 2);
            if (values.length == 2) {
                String username = values[0];
                String password = values[1];
                
                if (authenticate(username, password)) {
                    // 인증 성공
                    chain.doFilter(request, response);
                    return;
                }
            }
        }
        
        // 인증 실패
        resp.setHeader("WWW-Authenticate", "Basic realm=\"Restricted Area\"");
        resp.sendError(HttpServletResponse.SC_UNAUTHORIZED);
    }
    
    private boolean authenticate(String username, String password) {
        String storedPassword = USERS.get(username);
        return storedPassword != null && storedPassword.equals(password);
    }
    
    @Override
    public void init(FilterConfig filterConfig) throws ServletException {}
    
    @Override
    public void destroy() {}
}
```

### 보안 강화된 Basic Auth (암호 해시 사용)
```java
public class SecureBasicAuthFilter implements Filter {
    
    // 해시된 패스워드 저장 (실제로는 데이터베이스 사용)
    private static final Map<String, String> USER_HASHES = new HashMap<>();
    static {
        // "password123"의 bcrypt 해시
        USER_HASHES.put("admin", "$2a$10$N9qo8uLOickgx2ZMRZoMye3Z6g7z7Z7z7Z7z7Z7z7Z7z7Z7z7Z7");
        // "userpass"의 bcrypt 해시
        USER_HASHES.put("user", "$2a$10$N9qo8uLOickgx2ZMRZoMye3Z6g7z7Z7z7Z7z7Z7z7Z7z7Z7z7Z");
    }
    
    @Override
    public void doFilter(ServletRequest request, ServletResponse response, 
                         FilterChain chain) throws IOException, ServletException {
        
        HttpServletRequest req = (HttpServletRequest) request;
        HttpServletResponse resp = (HttpServletResponse) response;
        
        String authHeader = req.getHeader("Authorization");
        
        if (authHeader != null && authHeader.startsWith("Basic ")) {
            String base64Credentials = authHeader.substring("Basic ".length());
            String credentials = new String(Base64.getDecoder().decode(base64Credentials), 
                                          StandardCharsets.UTF_8);
            
            final String[] values = credentials.split(":", 2);
            if (values.length == 2) {
                String username = values[0];
                String password = values[1];
                
                if (authenticateSecure(username, password)) {
                    chain.doFilter(request, response);
                    return;
                }
            }
        }
        
        resp.setHeader("WWW-Authenticate", "Basic realm=\"Secure Area\"");
        resp.sendError(HttpServletResponse.SC_UNAUTHORIZED);
    }
    
    private boolean authenticateSecure(String username, String password) {
        String storedHash = USER_HASHES.get(username);
        if (storedHash == null) {
            return false;
        }
        
        // BCrypt를 사용한 패스워드 검증 (의존성 추가 필요)
        // return BCrypt.checkpw(password, storedHash);
        
        // 간단한 구현을 위해 평문 비교 (실제로는 사용하지 마세요)
        return storedHash.equals(hashPassword(password));
    }
    
    private String hashPassword(String password) {
        // 실제로는 BCrypt 등의 안전한 해시 함수 사용
        try {
            MessageDigest digest = MessageDigest.getInstance("SHA-256");
            byte[] hash = digest.digest(password.getBytes(StandardCharsets.UTF_8));
            return Base64.getEncoder().encodeToString(hash);
        } catch (NoSuchAlgorithmException e) {
            throw new RuntimeException(e);
        }
    }
    
    @Override
    public void init(FilterConfig filterConfig) throws ServletException {}
    
    @Override
    public void destroy() {}
}
```

## 3. 테스트 방법

### curl을 이용한 테스트
```bash
# 인증 없이 접근 시도
curl -X GET http://localhost:8080/api/users

# Basic Auth로 접근
curl -X GET -u admin:password123 http://localhost:8080/api/users

# POST 요청
curl -X POST -u admin:password123 \
  -H "Content-Type: application/json" \
  -d '{"name":"John Doe","email":"john@example.com"}' \
  http://localhost:8080/api/users
```

## 4. 주의사항

1. **보안**: Basic Auth는 base64 인코딩만 사용하므로 HTTPS와 함께 사용해야 합니다.
2. **패스워드 저장**: 실제 프로덕션 환경에서는 암호를 평문으로 저장하지 말고 bcrypt 등의 안전한 해시 함수를 사용하세요.
3. **세션 관리**: Basic Auth는 stateless하지만, 더 복잡한 인증 요구사항에는 JWT 등을 고려하세요.
4. **에러 처리**: 적절한 HTTP 상태 코드와 에러 메시지를 반환하도록 구현하세요.

이 구현은 순수 서블릿 API를 사용한 기본적인 REST API와 Basic Auth의 예시입니다. 실제 프로젝트에서는 Spring Boot나 JAX-RS 같은 프레임워크 사용을 고려하는 것이 좋습니다.
