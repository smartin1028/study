# Spring Boot와 Elasticsearch를 이용한 로그 파일 등록 및 조회 시스템

Spring Boot와 Elasticsearch를 사용하여 로그 파일 정보를 등록하고 조회하는 시스템을 구축하는 방법을 상세히 설명드리겠습니다.

## 1. 프로젝트 설정

### build.gradle 의존성 추가

```gradle
dependencies {
    implementation 'org.springframework.boot:spring-boot-starter-data-elasticsearch'
    implementation 'org.springframework.boot:spring-boot-starter-web'
    implementation 'org.springframework.boot:spring-boot-starter-thymeleaf'
    implementation 'com.fasterxml.jackson.core:jackson-databind'
    developmentOnly 'org.springframework.boot:spring-boot-devtools'

```

### application.yml 설정

```yaml
spring:
  data:
    elasticsearch:
      cluster-name: docker-cluster
      cluster-nodes: localhost:9200
      repositories:
        enabled: true

server:
  port: 8080

logging:
  level:
    org.springframework.data.elasticsearch: DEBUG
```

## 2. 엔티티 클래스 생성

```java
package com.example.logsystem.domain;

import org.springframework.data.annotation.Id;
import org.springframework.data.elasticsearch.annotations.Document;
import org.springframework.data.elasticsearch.annotations.Field;
import org.springframework.data.elasticsearch.annotations.FieldType;

import java.time.LocalDateTime;

@Document(indexName = "log_files")
public class LogFile {

    @Id
    private String id;

    @Field(type = FieldType.Text, name = "file_name")
    private String fileName;

    @Field(type = FieldType.Text, name = "file_path")
    private String filePath;

    @Field(type = FieldType.Long, name = "file_size")
    private Long fileSize;

    @Field(type = FieldType.Date, name = "created_date")
    private LocalDateTime createdDate;

    @Field(type = FieldType.Date, name = "modified_date")
    private LocalDateTime modifiedDate;

    @Field(type = FieldType.Text, name = "log_content")
    private String logContent;

    @Field(type = FieldType.Keyword, name = "log_level")
    private String logLevel;

    @Field(type = FieldType.Keyword, name = "application_name")
    private String applicationName;

    // 생성자, getter, setter
    public LogFile() {}

    public LogFile(String fileName, String filePath, Long fileSize, 
                  String logContent, String logLevel, String applicationName) {
        this.fileName = fileName;
        this.filePath = filePath;
        this.fileSize = fileSize;
        this.createdDate = LocalDateTime.now();
        this.modifiedDate = LocalDateTime.now();
        this.logContent = logContent;
        this.logLevel = logLevel;
        this.applicationName = applicationName;
    }

    // getter와 setter 메서드들...
}
```

package com.example.logsystem.repository;

import com.example.logsystem.domain.LogFile;
import org.springframework.data.elasticsearch.repository.ElasticsearchRepository;

@Repository
public interface LogFileRepository extends ElasticsearchRepository<LogFile, String> {
    
    List<LogFile> findByFileNameContaining(String fileName);
    List<LogFile> findByLogLevel(String logLevel);
    List<LogFile> findByApplicationName(String applicationName);
    List<LogFile> findByLogContentContaining(String keyword);
}

## 4. 서비스 클래스 생성

```java
package com.example.logsystem.service;

import com.example.logsystem.domain.LogFile;
import com.example.logsystem.repository.LogFileRepository;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.data.domain.Page;
import org.springframework.data.domain.Pageable;
import org.springframework.stereotype.Service;

import java.util.List;
import java.util.Optional;

@Service
public class LogFileService {

    @Autowired
    private LogFileRepository logFileRepository;

    public LogFile saveLogFile(LogFile logFile) {
        return logFileRepository.save(logFile);
    }

    public Optional<LogFile> findById(String id) {
        return logFileRepository.findById(id);
    }

    public List<LogFile> findAll() {
        return (List<LogFile>) logFileRepository.findAll();
    }

    public List<LogFile> findByFileName(String fileName) {
        return logFileRepository.findByFileNameContaining(fileName);
    }

    public List<LogFile> findByLogLevel(String logLevel) {
        return logFileRepository.findByLogLevel(logLevel);
    }

    public List<LogFile> searchByContent(String keyword) {
        return logFileRepository.findByLogContentContaining(keyword);
    }

    public void deleteLogFile(String id) {
        logFileRepository.deleteById(id);
    }
}
```

## 5. 컨트롤러 생성

```java
package com.example.logsystem.controller;

import com.example.logsystem.domain.LogFile;
import com.example.logsystem.service.LogFileService;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Controller;
import org.springframework.ui.Model;
import org.springframework.web.bind.annotation.*;

import java.util.List;

@Controller
@RequestMapping("/logs")
public class LogFileController {

    @Autowired
    private LogFileService logFileService;

    // 로그 등록 폼
    @GetMapping("/register")
    public String showRegisterForm(Model model) {
        model.addAttribute("logFile", new LogFile());
        return "log-register";
    }

    // 로그 등록 처리
    @PostMapping("/register")
    public String registerLogFile(@ModelAttribute LogFile logFile) {
        logFileService.saveLogFile(logFile);
        return "redirect:/logs/list";
    }

    // 로그 목록 조회
    @GetMapping("/list")
    public String listLogFiles(Model model) {
        List<LogFile> logFiles = logFileService.findAll();
        model.addAttribute("logFiles", logFiles);
        return "log-list";
    }

    // 로그 상세 조회
    @GetMapping("/detail/{id}")
    public String viewLogDetail(@PathVariable String id, Model model) {
        LogFile logFile = logFileService.findById(id)
                .orElseThrow(() -> new IllegalArgumentException("Invalid log id: " + id));
        model.addAttribute("logFile", logFile);
        return "log-detail";
    }

    // 로그 검색
    @GetMapping("/search")
    public String searchLogs(@RequestParam(required = false) String fileName,
                            @RequestParam(required = false) String logLevel,
                            @RequestParam(required = false) String keyword,
                            Model model) {
        List<LogFile> searchResults;
        
        if (fileName != null && !fileName.isEmpty()) {
            searchResults = logFileService.findByFileName(fileName);
        } else if (logLevel != null && !logLevel.isEmpty()) {
            searchResults = logFileService.findByLogLevel(logLevel);
        } else if (keyword != null && !keyword.isEmpty()) {
            searchResults = logFileService.searchByContent(keyword);
        } else {
            searchResults = logFileService.findAll();
        }
        
        model.addAttribute("logFiles", searchResults);
        return "log-list";
    }

    // 로그 삭제
    @PostMapping("/delete/{id}")
    public String deleteLogFile(@PathVariable String id) {
        logFileService.deleteLogFile(id);
        return "redirect:/logs/list";
    }
}
```

## 7. Elasticsearch 설정 클래스

```java
package com.example.logsystem.config;

import org.elasticsearch.client.RestHighLevelClient;
import org.springframework.context.annotation.Configuration;
import org.springframework.data.elasticsearch.client.ClientConfiguration;
import org.springframework.data.elasticsearch.client.RestClients;
import org.springframework.data.elasticsearch.config.AbstractElasticsearchConfiguration;
import org.springframework.data.elasticsearch.repository.config.EnableElasticsearchRepositories;

@Configuration
@EnableElasticsearchRepositories(basePackages = "com.example.logsystem.repository")
public class ElasticsearchConfig extends AbstractElasticsearchConfiguration {

    @Override
    public RestHighLevelClient elasticsearchClient() {
        ClientConfiguration clientConfiguration = ClientConfiguration.builder()
                .connectedTo("localhost:9200")
                .build();
        return RestClients.create(clientConfiguration).rest();
    }
}
```
