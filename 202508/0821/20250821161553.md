# Spring Boot와 Elasticsearch를 이용한 로그 등록 및 조회 시스템

Spring Boot와 Elasticsearch를 사용하여 로그 파일을 등록하고 조회하는 시스템을 구축하는 방법을 상세히 설명드리겠습니다.

## 2. Elasticsearch 도메인 모델

### LogDocument.java

```java
package com.example.logsearch.model;

import lombok.AllArgsConstructor;
import lombok.Data;
import lombok.NoArgsConstructor;
import org.springframework.data.annotation.Id;
import org.springframework.data.elasticsearch.annotations.Document;
import org.springframework.data.elasticsearch.annotations.Field;
import org.springframework.data.elasticsearch.annotations.FieldType;

import java.time.LocalDateTime;

@Data
@NoArgsConstructor
@AllArgsConstructor
@Document(indexName = "logs")
public class LogDocument {
    
    @Id
    private String id;
    
    @Field(type = FieldType.Date, pattern = "yyyy-MM-dd'T'HH:mm:ss.SSS")
    private LocalDateTime timestamp;
    
    @Field(type = FieldType.Keyword)
    private String level;
    
    @Field(type = FieldType.Text)
    private String message;
    
    @Field(type = FieldType.Keyword)
    private String loggerName;
    
    @Field(type = FieldType.Keyword)
    private String threadName;
    
    @Field(type = FieldType.Keyword)
    private String sourceFile;
    
    @Field(type = FieldType.Integer)
    private int lineNumber;
    
    @Field(type = FieldType.Keyword)
    private String application;
}
```

package com.example.logsearch.repository;

import com.example.logsearch.model.LogDocument;
import org.springframework.data.domain.Page;
import org.springframework.data.domain.Pageable;
import org.springframework.data.elasticsearch.repository.ElasticsearchRepository;
import org.springframework.stereotype.Repository;

import java.time.LocalDateTime;
import java.util.List;

@Repository
public interface LogRepository extends ElasticsearchRepository<LogDocument, String> {
    
    List<LogDocument> findByLevel(String level);
    
    List<LogDocument> findByApplication(String application);
    
    List<LogDocument> findByTimestampBetween(LocalDateTime start, LocalDateTime end);
    
    Page<LogDocument> findByMessageContaining(String keyword, Pageable pageable);
    
    List<LogDocument> findByLevelAndTimestampBetween(String level, LocalDateTime start, LocalDateTime end);
}

## 4. 로그 파일 파서

### LogParser.java

```java
package com.example.logsearch.service;

import com.example.logsearch.model.LogDocument;
import org.apache.commons.io.FileUtils;
import org.apache.commons.io.LineIterator;
import org.springframework.stereotype.Component;

import java.io.File;
import java.io.IOException;
import java.time.LocalDateTime;
import java.time.format.DateTimeFormatter;
import java.util.ArrayList;
import java.util.List;
import java.util.regex.Matcher;
import java.util.regex.Pattern;

@Component
public class LogParser {
    
    private static final Pattern LOG_PATTERN = Pattern.compile(
        "(\\d{4}-\\d{2}-\\d{2} \\d{2}:\\d{2}:\\d{2}\\.\\d{3}) (\\w+) \\[(.*?)\\] (.*?) - (.*)"
    );
    
    private static final DateTimeFormatter FORMATTER = 
        DateTimeFormatter.ofPattern("yyyy-MM-dd HH:mm:ss.SSS");
    
    public List<LogDocument> parseLogFile(String filePath, String applicationName) throws IOException {
        List<LogDocument> logs = new ArrayList<>();
        File file = new File(filePath);
        
        if (!file.exists()) {
            throw new IOException("File not found: " + filePath);
        }
        
        LineIterator iterator = FileUtils.lineIterator(file, "UTF-8");
        try {
            while (iterator.hasNext()) {
                String line = iterator.nextLine();
                LogDocument log = parseLine(line, applicationName);
                if (log != null) {
                    logs.add(log);
                }
            }
        } finally {
            LineIterator.closeQuietly(iterator);
        }
        
        return logs;
    }
    
    private LogDocument parseLine(String line, String applicationName) {
        Matcher matcher = LOG_PATTERN.matcher(line);
        
        if (matcher.find()) {
            try {
                LocalDateTime timestamp = LocalDateTime.parse(matcher.group(1), FORMATTER);
                String level = matcher.group(2);
                String threadName = matcher.group(3);
                String loggerName = matcher.group(4);
                String message = matcher.group(5);
                
                return new LogDocument(
                    null,
                    timestamp,
                    level,
                    message,
                    loggerName,
                    threadName,
                    extractSourceFile(loggerName),
                    extractLineNumber(message),
                    applicationName
                );
            } catch (Exception e) {
                System.err.println("Failed to parse line: " + line);
                return null;
            }
        }
        return null;
    }
    
    private String extractSourceFile(String loggerName) {
        if (loggerName != null && loggerName.contains(".")) {
            String[] parts = loggerName.split("\\.");
            return parts[parts.length - 1] + ".java";
        }
        return "Unknown";
    }
    
    private int extractLineNumber(String message) {
        Pattern pattern = Pattern.compile("\\(.*?:(\\d+)\\)");
        Matcher matcher = pattern.matcher(message);
        if (matcher.find()) {
            return Integer.parseInt(matcher.group(1));
        }
        return 0;
    }
}
```

## 5. 서비스 레이어

### LogService.java

```java
package com.example.logsearch.service;

import com.example.logsearch.model.LogDocument;
import com.example.logsearch.repository.LogRepository;
import lombok.RequiredArgsConstructor;
import org.springframework.data.domain.Page;
import org.springframework.data.domain.Pageable;
import org.springframework.stereotype.Service;

import java.io.IOException;
import java.time.LocalDateTime;
import java.util.List;

@Service
@RequiredArgsConstructor
public class LogService {
    
    private final LogRepository logRepository;
    private final LogParser logParser;
    
    public void importLogFile(String filePath, String applicationName) throws IOException {
        List<LogDocument> logs = logParser.parseLogFile(filePath, applicationName);
        logRepository.saveAll(logs);
    }
    
    public List<LogDocument> getAllLogs() {
        return (List<LogDocument>) logRepository.findAll();
    }
    
    public Page<LogDocument> searchLogs(String keyword, Pageable pageable) {
        return logRepository.findByMessageContaining(keyword, pageable);
    }
    
    public List<LogDocument> getLogsByLevel(String level) {
        return logRepository.findByLevel(level);
    }
    
    public List<LogDocument> getLogsByApplication(String application) {
        return logRepository.findByApplication(application);
    }
    
    public List<LogDocument> getLogsByTimeRange(LocalDateTime start, LocalDateTime end) {
        return logRepository.findByTimestampBetween(start, end);
    }
    
    public List<LogDocument> getLogsByLevelAndTimeRange(String level, LocalDateTime start, LocalDateTime end) {
        return logRepository.findByLevelAndTimestampBetween(level, start, end);
    }
    
    public void deleteAllLogs() {
        logRepository.deleteAll();
    }
}
```

## 6. 컨트롤러

### LogController.java

```java
package com.example.logsearch.controller;

import com.example.logsearch.model.LogDocument;
import com.example.logsearch.service.LogService;
import lombok.RequiredArgsConstructor;
import org.springframework.data.domain.Page;
import org.springframework.data.domain.PageRequest;
import org.springframework.data.domain.Pageable;
import org.springframework.format.annotation.DateTimeFormat;
import org.springframework.stereotype.Controller;
import org.springframework.ui.Model;
import org.springframework.web.bind.annotation.*;
import org.springframework.web.multipart.MultipartFile;

import java.io.IOException;
import java.nio.file.Files;
import java.nio.file.Path;
import java.nio.file.Paths;
import java.time.LocalDateTime;
import java.util.List;

@Controller
@RequestMapping("/logs")
@RequiredArgsConstructor
public class LogController {
    
    private final LogService logService;
    
    private static final String UPLOAD_DIR = "./uploads/";
    
    @GetMapping
    public String showLogs(Model model,
                          @RequestParam(defaultValue = "0") int page,
                          @RequestParam(defaultValue = "20") int size) {
        Pageable pageable = PageRequest.of(page, size);
        Page<LogDocument> logPage = logService.searchLogs("", pageable);
        
        model.addAttribute("logs", logPage.getContent());
        model.addAttribute("currentPage", page);
        model.addAttribute("totalPages", logPage.getTotalPages());
        model.addAttribute("totalItems", logPage.getTotalElements());
        
        return "logs/list";
    }
    
    @GetMapping("/upload")
    public String showUploadForm() {
        return "logs/upload";
    }
    
    @PostMapping("/upload")
    public String uploadLogFile(@RequestParam("file") MultipartFile file,
                               @RequestParam("applicationName") String applicationName,
                               Model model) {
        if (file.isEmpty()) {
            model.addAttribute("error", "Please select a file to upload");
            return "logs/upload";
        }
        
        try {
            // Create upload directory if it doesn't exist
            Path uploadPath = Paths.get(UPLOAD_DIR);
            if (!Files.exists(uploadPath)) {
                Files.createDirectories(uploadPath);
            }
            
            // Save file
            Path filePath = uploadPath.resolve(file.getOriginalFilename());
            Files.write(filePath, file.getBytes());
            
            // Import logs to Elasticsearch
            logService.importLogFile(filePath.toString(), applicationName);
            
            model.addAttribute("message", "File uploaded and logs imported successfully!");
            
        } catch (IOException e) {
            model.addAttribute("error", "Failed to upload file: " + e.getMessage());
        }
        
        return "logs/upload";
    }
    
    @GetMapping("/search")
    public String searchLogs(@RequestParam("keyword") String keyword,
                            @RequestParam(defaultValue = "0") int page,
                            @RequestParam(defaultValue = "20") int size,
                            Model model) {
        Pageable pageable = PageRequest.of(page, size);
        Page<LogDocument> logPage = logService.searchLogs(keyword, pageable);
        
        model.addAttribute("logs", logPage.getContent());
        model.addAttribute("keyword", keyword);
        model.addAttribute("currentPage", page);
        model.addAttribute("totalPages", logPage.getTotalPages());
        model.addAttribute("totalItems", logPage.getTotalElements());
        
        return "logs/list";
    }
    
    @GetMapping("/filter")
    public String filterLogs(@RequestParam(value = "level", required = false) String level,
                            @RequestParam(value = "application", required = false) String application,
                            @RequestParam(value = "startDate", required = false) 
                            @DateTimeFormat(iso = DateTimeFormat.ISO.DATE_TIME) LocalDateTime startDate,
                            @RequestParam(value = "endDate", required = false) 
                            @DateTimeFormat(iso = DateTimeFormat.ISO.DATE_TIME) LocalDateTime endDate,
                            Model model) {
        
        List<LogDocument> logs;
        
        if (level != null && startDate != null && endDate != null) {
            logs = logService.getLogsByLevelAndTimeRange(level, startDate, endDate);
        } else if (level != null) {
            logs = logService.getLogsByLevel(level);
        } else if (application != null) {
            logs = logService.getLogsByApplication(application);
        } else if (startDate != null && endDate != null) {
            logs = logService.getLogsByTimeRange(startDate, endDate);
        } else {
            logs = logService.getAllLogs();
        }
        
        model.addAttribute("logs", logs);
        model.addAttribute("level", level);
        model.addAttribute("application", application);
        model.addAttribute("startDate", startDate);
        model.addAttribute("endDate", endDate);
        
        return "logs/filter";
    }
    
    @PostMapping("/delete-all")
    public String deleteAllLogs(Model model) {
        logService.deleteAllLogs();
        model.addAttribute("message", "All logs deleted successfully!");
        return "redirect:/logs";
    }
}
```
