## 5. 리스트 (List)

## 7. 반복문 (for, while)

```python
# for 문
fruits = ["apple", "banana", "cherry"]
for fruit in fruits:
    print(fruit)

# range() 함수 사용
for i in range(5):        # 0부터 4까지
    print(i)

for i in range(2, 6):     # 2부터 5까지
    print(i)

# while 문
count = 0
while count < 5:
    print(count)
    count += 1

# break와 continue
for i in range(10):
    if i == 3:
        continue  # 다음 반복으로 건너뜀
    if i == 7:
        break     # 반복문 종료
    print(i)
```

## 8. 함수 정의

```python
# 기본 함수
def greet(name):
    return f"Hello, {name}!"

print(greet("Alice"))  # "Hello, Alice!"

# 기본 매개변수
def introduce(name, age=30):
    return f"I'm {name}, {age} years old"

print(introduce("Bob"))      # "I'm Bob, 30 years old"
print(introduce("Charlie", 25))  # "I'm Charlie, 25 years old"

# 가변 인자
def sum_all(*args):
    return sum(args)

print(sum_all(1, 2, 3, 4, 5))  # 15

# 키워드 가변 인자
def print_info(**kwargs):
    for key, value in kwargs.items():
        print(f"{key}: {value}")

print_info(name="John", age=25, city="Seoul")
```

## 9. 딕셔너리 (Dictionary)

```python
# 딕셔너리 생성
person = {
    "name": "John",
    "age": 30,
    "city": "New York"
}

# 값 접근
print(person["name"])    # "John"
print(person.get("age")) # 30

# 값 추가/수정
person["job"] = "Engineer"  # 추가
person["age"] = 31         # 수정

# 키-값 쌍 순회
for key, value in person.items():
    print(f"{key}: {value}")

# 딕셔너리 컴프리헨션
squares = {x: x**2 for x in range(5)}  # {0: 0, 1: 1, 2: 4, 3: 9, 4: 16}
```

## 11. 클래스와 객체지향 프로그래밍

```python
class Person:
    # 클래스 변수
    species = "Homo sapiens"
    
    # 생성자
    def __init__(self, name, age):
        self.name = name  # 인스턴스 변수
        self.age = age
    
    # 인스턴스 메서드
    def introduce(self):
        return f"My name is {self.name} and I'm {self.age} years old"
    
    # 클래스 메서드
    @classmethod
    def get_species(cls):
        return cls.species

# 객체 생성
person1 = Person("Alice", 25)
print(person1.introduce())  # "My name is Alice and I'm 25 years old"
print(Person.get_species()) # "Homo sapiens"
```
