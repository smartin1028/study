### 1. 정렬 알고리즘 (Sorting Algorithms)

```python
# 버블 정렬
def bubble_sort(arr):
    n = len(arr)
    for i in range(n):
        for j in range(0, n-i-1):
            if arr[j] > arr[j+1]:
                arr[j], arr[j+1] = arr[j+1], arr[j]
    return arr

# 퀵 정렬
def quick_sort(arr):
    if len(arr) <= 1:
        return arr
    pivot = arr[len(arr)//2]
    left = [x for x in arr if x < pivot]
    middle = [x for x in arr if x == pivot]
    right = [x for x in arr if x > pivot]
    return quick_sort(left) + middle + quick_sort(right)
```

### 2. 검색 알고리즘 (Searching Algorithms)
```python
# 이진 검색
def binary_search(arr, target):
    low, high = 0, len(arr)-1
    while low <= high:
        mid = (low + high) // 2
        if arr[mid] == target:
            return mid
        elif arr[mid] < target:
            low = mid + 1
        else:
            high = mid - 1
    return -1
```

### 3. 그래프 알고리즘 (Graph Algorithms)
```python
from collections import deque, defaultdict

# BFS (너비 우선 탐색)
def bfs(graph, start):
    visited = set()
    queue = deque([start])
    result = []
    
    while queue:
        node = queue.popleft()
        if node not in visited:
            visited.add(node)
            result.append(node)
            queue.extend(graph[node])
    return result

# 다익스트라 알고리즘 (최단 경로)
import heapq

def dijkstra(graph, start):
    distances = {node: float('inf') for node in graph}
    distances[start] = 0
    queue = [(0, start)]
    
    while queue:
        current_dist, current_node = heapq.heappop(queue)
        
        if current_dist > distances[current_node]:
            continue
            
        for neighbor, weight in graph[current_node].items():
            distance = current_dist + weight
            if distance < distances[neighbor]:
                distances[neighbor] = distance
                heapq.heappush(queue, (distance, neighbor))
    
    return distances
```

### 4. 머신러닝 알고리즘
```python
# K-평균 클러스터링
from sklearn.cluster import KMeans
import numpy as np

# 간단한 구현 예시
def simple_kmeans(data, k, max_iters=100):
    # 중심점 랜덤 초기화
    centroids = data[np.random.choice(range(len(data)), k, replace=False)]
    
    for _ in range(max_iters):
        # 각 점을 가장 가까운 중심점에 할당
        labels = np.argmin(np.linalg.norm(data[:, np.newaxis] - centroids, axis=2), axis=1)
        
        # 새로운 중심점 계산
        new_centroids = np.array([data[labels == i].mean(axis=0) for i in range(k)])
        
        # 수렴 확인
        if np.all(centroids == new_centroids):
            break
        centroids = new_centroids
    
    return labels, centroids
```

### 6. 데이터 압축 알고리즘
```python
# 허프만 코딩 (Huffman Coding)
from heapq import heappush, heappop
from collections import defaultdict

class HuffmanNode:
    def __init__(self, char, freq):
        self.char = char
        self.freq = freq
        self.left = None
        self.right = None

def build_huffman_tree(text):
    # 빈도수 계산
    frequency = defaultdict(int)
    for char in text:
        frequency[char] += 1
    
    # 최소 힙 구성
    heap = []
    for char, freq in frequency.items():
        heappush(heap, (freq, HuffmanNode(char, freq)))
    
    # 트리 구성
    while len(heap) > 1:
        freq1, node1 = heappop(heap)
        freq2, node2 = heappop(heap)
        merged = HuffmanNode(None, freq1 + freq2)
        merged.left = node1
        merged.right = node2
        heappush(heap, (merged.freq, merged))
    
    return heappop(heap)[1]
```

### 7. 웹 크롤러 알고리즘
```python
import requests
from bs4 import BeautifulSoup
from urllib.parse import urljoin, urlparse
from collections import deque

def web_crawler(start_url, max_pages=100):
    visited = set()
    queue = deque([start_url])
    results = []
    
    while queue and len(visited) < max_pages:
        url = queue.popleft()
        
        if url in visited:
            continue
            
        try:
            response = requests.get(url, timeout=5)
            soup = BeautifulSoup(response.text, 'html.parser')
            
            # 데이터 추출
            title = soup.title.string if soup.title else 'No title'
            results.append({'url': url, 'title': title})
            visited.add(url)
            
            # 링크 수집
            for link in soup.find_all('a', href=True):
                absolute_url = urljoin(url, link['href'])
                if urlparse(absolute_url).netloc == urlparse(start_url).netloc:
                    queue.append(absolute_url)
                    
        except Exception as e:
            print(f"Error crawling {url}: {e}")
    
    return results
```

### 8. 추천 시스템 알고리즘
```python
import numpy as np
from sklearn.metrics.pairwise import cosine_similarity

# 협업 필터링 기반 추천 시스템
def collaborative_filtering(user_item_matrix, user_id, n_recommendations=5):
    # 사용자 간 유사도 계산
    user_similarity = cosine_similarity(user_item_matrix)
    
    # 타겟 사용자의 유사한 사용자 찾기
    similar_users = np.argsort(user_similarity[user_id])[::-1][1:]
    
    # 추천 점수 계산
    recommendations = {}
    for similar_user in similar_users:
        for item in range(user_item_matrix.shape[1]):
            if user_item_matrix[user_id, item] == 0 and user_item_matrix[similar_user, item] > 0:
                if item not in recommendations:
                    recommendations[item] = 0
                recommendations[item] += user_similarity[user_id, similar_user] * user_item_matrix[similar_user, item]
    
    # 상위 N개 추천 반환
    return sorted(recommendations.items(), key=lambda x: x[1], reverse=True)[:n_recommendations]
```
