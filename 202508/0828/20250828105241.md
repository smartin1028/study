# 윈도우에서 Spring Boot 자동 실행 스크립트 생성 방법

## 배치 파일(.bat) 생성하기

다음 내용으로 `start-springboot.bat` 파일을 생성하세요:

```batch
@echo off
chcp 65001 >nul
setlocal enabledelayedexpansion

REM 애플리케이션 정보 설정
set APP_NAME=your-application-name
set JAR_FILE=your-application.jar
set JAVA_OPTS=-Xmx512m -Xms256m
set SPRING_OPTS=--spring.profiles.active=dev

REM JAR 파일 존재 여부 확인
if not exist "%JAR_FILE%" (
    echo Error: %JAR_FILE% 파일을 찾을 수 없습니다.
    pause
    exit /b 1
)

REM Java 설치 여부 확인
java -version >nul 2>&1
if errorlevel 1 (
    echo Error: Java가 설치되어 있지 않거나 PATH에 등록되어 있지 않습니다.
    pause
    exit /b 1
)

echo ========================================
echo Spring Boot 애플리케이션 시작
echo 애플리케이션: %APP_NAME%
echo JAR 파일: %JAR_FILE%
echo ========================================

REM 애플리케이션 실행
java %JAVA_OPTS% -jar "%JAR_FILE%" %SPRING_OPTS%

REM 실행 후 일시정지 (오류 확인용)
if errorlevel 1 (
    echo.
    echo 애플리케이션 실행 중 오류가 발생했습니다.
    pause
) else (
    echo.
    echo 애플리케이션이 정상적으로 종료되었습니다.
)

endlocal
```

## 고급 버전 스크립트 (로그 관리 포함)

```batch
@echo off
chcp 65001 >nul
setlocal enabledelayedexpansion

REM 설정 변수
set APP_NAME=my-springboot-app
set JAR_FILE=app.jar
set LOG_DIR=logs
set JAVA_HOME=C:\Program Files\Java\jdk-17
set JAVA_OPTS=-Xmx512m -Xms256m -server -Dfile.encoding=UTF-8
set SPRING_OPTS=--spring.profiles.active=dev --logging.file.name=%LOG_DIR%\%APP_NAME%.log

REM 로그 디렉토리 생성
if not exist "%LOG_DIR%" mkdir "%LOG_DIR%"

REM JAR 파일 확인
if not exist "%JAR_FILE%" (
    echo [ERROR] %JAR_FILE% 파일을 찾을 수 없습니다.
    echo 사용 가능한 JAR 파일:
    dir *.jar /b
    pause
    exit /b 1
)

REM Java 확인
if defined JAVA_HOME (
    set "JAVA_EXE=%JAVA_HOME%\bin\java.exe"
) else (
    set "JAVA_EXE=java"
)

if not exist "!JAVA_EXE!" (
    echo [ERROR] Java를 찾을 수 없습니다. JAVA_HOME을 설정하거나 PATH에 Java를 추가하세요.
    pause
    exit /b 1
)

echo ========================================
echo %APP_NAME% 시작
echo 시간: %date% %time%
echo JAR: %JAR_FILE%
echo ========================================

REM PID 파일을 위한 타임스탬프 생성
for /f "tokens=2 delims==" %%I in ('wmic os get localdatetime /format:list') do set datetime=%%I
set timestamp=%datetime:~0,4%-%datetime:~4,2%-%datetime:~6,2%_%datetime:~8,2%-%datetime:~10,2%-%datetime:~12,2%

REM 애플리케이션 실행
echo 애플리케이션을 시작합니다...
"!JAVA_EXE!" %JAVA_OPTS% -jar "%JAR_FILE%" %SPRING_OPTS%

set EXIT_CODE=%errorlevel%

echo.
if !EXIT_CODE! equ 0 (
    echo [SUCCESS] 애플리케이션이 정상 종료되었습니다.
) else (
    echo [ERROR] 애플리케이션 종료 코드: !EXIT_CODE!
    echo 최근 로그 내용:
    if exist "%LOG_DIR%\%APP_NAME%.log" (
        tail -10 "%LOG_DIR%\%APP_NAME%.log"
    )
)

pause
endlocal
```

## 자동 재시작 스크립트

```batch
@echo off
chcp 65001 >nul
setlocal enabledelayedexpansion

set APP_NAME=my-app
set JAR_FILE=app.jar
set MAX_RESTARTS=5
set RESTART_DELAY=10

set RESTART_COUNT=0

:start_app
echo ========================================
echo %APP_NAME% 시작 (시도: !RESTART_COUNT!/%MAX_RESTARTS%)
echo 시간: %date% %time%
echo ========================================

java -jar "%JAR_FILE%"

set EXIT_CODE=%errorlevel%
set /a RESTART_COUNT+=1

if !EXIT_CODE! equ 0 (
    echo 애플리케이션이 정상 종료되었습니다.
    goto :end
)

if !RESTART_COUNT! gtr %MAX_RESTARTS% (
    echo 최대 재시작 횟수(%MAX_RESTARTS%)를 초과했습니다.
    goto :end
)

echo 애플리케이션이 비정상 종료되었습니다. !RESTART_DELAY!초 후 재시작합니다...
timeout /t %RESTART_DELAY% /nobreak >nul
goto :start_app

:end
echo 스크립트 종료
pause
```

## 사용 방법

1. **스크립트 수정**:
   - `APP_NAME`: 애플리케이션 이름 설정
   - `JAR_FILE`: 실제 JAR 파일명으로 변경
   - `JAVA_OPTS`, `SPRING_OPTS`: 필요에 따라 수정

2. **스크립트 실행**:
   ```cmd
   start-springboot.bat
   ```

3. **관리자 권한으로 실행** (필요한 경우):
   ```cmd
  右键点击 -> "관리자 권한으로 실행"
   ```

## 추가 팁

- **서비스로 등록**하려면 nssm(Non-Sucking Service Manager) 사용:
  ```cmd
  nssm install MySpringBootApp "C:\path\to\java.exe" "-jar C:\path\to\your-app.jar"
  ```

- **환경 변수 설정**은 `setx` 명령어나 시스템 속성에서 가능

- **로그 로테이션**은 logback이나 log4j2 설정 파일에서 구성 가능

스크립트는 실제 환경에 맞게 수정하여 사용하시기 바랍니다.
