# 윈도우에서 Spring Boot 자동 실행하는 CMD 스크립트

윈도우에서 Spring Boot 애플리케이션을 자동으로 실행하는 배치 스크립트를 생성하는 방법을 상세히 설명해 드리겠습니다.

## 기본 배치 스크립트 (start-springboot.bat)

```batch
@echo off
chcp 65001 >nul
setlocal enabledelayedexpansion

REM 프로젝트 정보 설정
set PROJECT_DIR=C:\path\to\your\springboot\project
set JAR_FILE=your-application.jar
set JAVA_OPTS=-Xmx512m -Xms256m

REM 로그 디렉토리 생성
if not exist "%PROJECT_DIR%\logs" mkdir "%PROJECT_DIR%\logs"

echo ========================================
echo Spring Boot Application 시작
echo ========================================
echo 시작 시간: %date% %time%
echo 프로젝트 디렉토리: %PROJECT_DIR%
echo ========================================

REM 디렉토리 이동
cd /d "%PROJECT_DIR%"

REM 기존 프로세스 종료 (선택사항)
REM taskkill /f /im java.exe 2>nul

REM 애플리케이션 실행
java %JAVA_OPTS% -jar "%JAR_FILE%" --spring.profiles.active=dev

echo ========================================
echo 애플리케이션이 종료되었습니다.
echo 종료 시간: %date% %time%
echo ========================================

pause
```

## 고급 버전 - 로그 관리 포함 (start-springboot-advanced.bat)

```batch
@echo off
chcp 65001 >nul
setlocal enabledelayedexpansion

REM 설정 변수
set PROJECT_DIR=C:\path\to\your\springboot\project
set JAR_FILE=target\your-application-0.0.1-SNAPSHOT.jar
set APP_NAME=SpringBootApp
set LOG_DIR=%PROJECT_DIR%\logs
set JAVA_HOME=C:\Program Files\Java\jdk-17
set JAVA_OPTS=-Xmx1024m -Xms512m -Dfile.encoding=UTF-8
set SPRING_OPTS=--spring.profiles.active=prod --server.port=8080

REM 로그 파일 설정
set LOG_FILE=%LOG_DIR%\%APP_NAME%_%date:~0,4%%date:~5,2%%date:~8,2%_%time:~0,2%%time:~3,2%.log

REM 디렉토리 생성
if not exist "%LOG_DIR%" mkdir "%LOG_DIR%"

echo ======================================== >> "%LOG_FILE%"
echo [%date% %time%] %APP_NAME% 시작 >> "%LOG_FILE%"
echo ======================================== >> "%LOG_FILE%"

REM JAVA_HOME 설정
if defined JAVA_HOME (
    set "PATH=%JAVA_HOME%\bin;%PATH%"
)

REM 디렉토리 이동
cd /d "%PROJECT_DIR%"

echo JAVA_HOME: %JAVA_HOME% >> "%LOG_FILE%"
echo 프로젝트 디렉토리: %PROJECT_DIR% >> "%LOG_FILE%"
echo JAR 파일: %JAR_FILE% >> "%LOG_FILE%"
echo ======================================== >> "%LOG_FILE%"

REM 애플리케이션 실행 (로그 파일에도 출력)
echo [%date% %time%] 애플리케이션 시작 중... >> "%LOG_FILE%"
java %JAVA_OPTS% -jar "%JAR_FILE%" %SPRING_OPTS% >> "%LOG_FILE%" 2>&1

REM 종료 코드 확인
if errorlevel 1 (
    echo [%date% %time%] 애플리케이션 비정상 종료 (에러 코드: !errorlevel!) >> "%LOG_FILE%"
) else (
    echo [%date% %time%] 애플리케이션 정상 종료 >> "%LOG_FILE%"
)

echo ======================================== >> "%LOG_FILE%"
echo [%date% %time%] %APP_NAME% 종료 >> "%LOG_FILE%"
echo ======================================== >> "%LOG_FILE%"

REM 콘솔에도 메시지 출력
echo.
echo 애플리케이션이 종료되었습니다.
echo 로그 파일: %LOG_FILE%
pause
```

## Maven/Gradle 빌드 포함 스크립트 (build-and-run.bat)

```batch
@echo off
chcp 65001 >nul
setlocal enabledelayedexpansion

set PROJECT_DIR=C:\path\to\your\springboot\project
set LOG_DIR=%PROJECT_DIR%\logs

REM 로그 디렉토리 생성
if not exist "%LOG_DIR%" mkdir "%LOG_DIR%"

echo ========================================
echo Spring Boot 빌드 및 실행
echo ========================================

cd /d "%PROJECT_DIR%"

REM Maven 빌드 (Maven 사용 시)
echo Maven 빌드 시작...
call mvn clean package -DskipTests

if errorlevel 1 (
    echo 빌드 실패!
    pause
    exit /b 1
)

echo 빌드 성공!

REM JAR 파일 찾기
for /f "delims=" %%i in ('dir /b /s target\*.jar') do set JAR_FILE=%%i

if not defined JAR_FILE (
    echo JAR 파일을 찾을 수 없습니다!
    pause
    exit /b 1
)

echo 실행 JAR: %JAR_FILE%

REM 애플리케이션 실행
echo 애플리케이션 시작...
java -jar "%JAR_FILE%" --spring.profiles.active=local

pause
```

## 서비스 형태로 실행하는 스크립트 (run-as-service.bat)

```batch
@echo off
setlocal enabledelayedexpansion

set PROJECT_DIR=C:\path\to\your\springboot\project
set JAR_FILE=your-application.jar
set SERVICE_NAME=SpringBootService
set LOG_FILE=%PROJECT_DIR%\service.log

echo Spring Boot 서비스 시작: %SERVICE_NAME%

REM nssm (Non-Sucking Service Manager) 사용 예제
REM nssm install %SERVICE_NAME% "%JAVA_HOME%\bin\java.exe" "-jar \"%PROJECT_DIR%\%JAR_FILE%\""
REM nssm start %SERVICE_NAME%

REM 또는 java servicemaker 같은 도구 사용

echo 서비스가 설치되었습니다.
echo 관리 명령어:
echo   - 시작: net start %SERVICE_NAME%
echo   - 중지: net stop %SERVICE_NAME%
echo   - 상태: sc query %SERVICE_NAME%

pause
```

## 사용 방법

1. **스크립트 생성**: 원하는 스크립트 내용을 `.bat` 파일로 저장합니다.
2. **변수 수정**: `PROJECT_DIR`, `JAR_FILE`, `JAVA_OPTS` 등을 실제 환경에 맞게 수정합니다.
3. **실행**: 더블클릭하거나 명령프롬프트에서 실행합니다.

## 추가 설정 팁

### 환경 변수 설정
```batch
set SPRING_PROFILES_ACTIVE=dev
set SERVER_PORT=8080
set JAVA_OPTS=-Xmx512m -Xms256m -Dspring.profiles.active=%SPRING_PROFILES_ACTIVE%
```

### 자동 재시작 스크립트 (restart-on-failure.bat)
```batch
@echo off
:restart
echo [%date% %time%] 애플리케이션 시작
call start-springboot.bat
echo [%date% %time%] 애플리케이션 종료, 10초 후 재시작...
timeout /t 10 /nobreak
goto restart
```

## 주의사항

1. **경로 설정**: 모든 경로는 실제 프로젝트 구조에 맞게 수정해야 합니다.
2. **Java 버전**: `JAVA_HOME`을 실제 JDK 경로로 설정하세요.
3. **권한**: 관리자 권한이 필요한 경우 스크립트를 관리자 모드로 실행하세요.
4. **인코딩**: `chcp 65001`은 UTF-8 인코딩을 위해 사용됩니다.

이 스크립트들을根据需要 수정하여 사용하시면 윈도우 환경에서 Spring Boot 애플리케이션을 편리하게 관리할 수 있습니다.
