# 파이썬 로그 파일 분석: 레벨 추출 및 메시지 구분

## 1. 로그 레벨을 정규식으로 추출하고 추가하는 방법

```python
import re
import pandas as pd

def extract_log_level(log_line):
    """
    로그 라인에서 로그 레벨을 추출하는 함수
    """
    # 일반적인 로그 레벨 패턴 (DEBUG, INFO, WARN, WARNING, ERROR, CRITICAL, FATAL)
    level_pattern = r'\b(DEBUG|INFO|WARN|WARNING|ERROR|CRITICAL|FATAL)\b'
    
    match = re.search(level_pattern, log_line, re.IGNORECASE)
    if match:
        return match.group(1).upper()  # 대문자로 통일
    else:
        return 'UNKNOWN'

# 예제 사용법
log_lines = [
    "2023-10-15 10:30:25 INFO User login successful",
    "2023-10-15 10:31:45 ERROR Database connection failed",
    "2023-10-15 10:32:15 WARN Disk space running low",
    "2023-10-15 10:33:00 DEBUG Processing request ID: 12345"
]

for line in log_lines:
    level = extract_log_level(line)
    print(f"로그: {line}")
    print(f"추출된 레벨: {level}")
    print("-" * 50)
```

## 2. 괄호 위치 기반 메시지 구분 방법

```python
def extract_message_with_brackets(log_line):
    """
    첫 번째 괄호 이후의 문자열을 메시지로 추출하는 함수
    """
    # 첫 번째 괄호의 위치 찾기
    bracket_pos = log_line.find('(')
    
    if bracket_pos != -1:
        # 괄호 이후의 모든 내용을 메시지로 추출
        message = log_line[bracket_pos:]
        return message
    else:
        # 괄호가 없는 경우 전체 라인 반환 또는 다른 처리
        return log_line

# 예제 사용법
log_with_brackets = [
    "2023-10-15 10:30:25 INFO (UserModule) User login successful for user: john_doe",
    "2023-10-15 10:31:45 ERROR (Database) Connection failed: timeout after 30 seconds",
    "2023-10-15 10:32:15 WARN (System) Disk space running low on /dev/sda1 (85% used)",
    "2023-10-15 10:33:00 DEBUG Processing request without brackets"
]

for line in log_with_brackets:
    message = extract_message_with_brackets(line)
    print(f"원본: {line}")
    print(f"메시지: {message}")
    print("-" * 70)
```

## 3. 통합 로그 분석기 클래스

```python
import re
from typing import List, Dict, Tuple

class LogAnalyzer:
    def __init__(self):
        self.level_pattern = r'\b(DEBUG|INFO|WARN|WARNING|ERROR|CRITICAL|FATAL)\b'
    
    def extract_log_info(self, log_line: str) -> Dict[str, str]:
        """
        로그 라인에서 레벨과 메시지를 추출
        """
        # 레벨 추출
        level_match = re.search(self.level_pattern, log_line, re.IGNORECASE)
        level = level_match.group(1).upper() if level_match else 'UNKNOWN'
        
        # 메시지 추출 (첫 번째 괄호 이후)
        bracket_pos = log_line.find('(')
        if bracket_pos != -1:
            message = log_line[bracket_pos:]
        else:
            # 괄호가 없으면 레벨 이후의 내용을 메시지로 사용
            if level_match:
                message_start = level_match.end()
                message = log_line[message_start:].strip()
            else:
                message = log_line
        
        return {
            'level': level,
            'message': message.strip(),
            'original': log_line
        }
    
    def analyze_log_file(self, file_path: str) -> List[Dict[str, str]]:
        """
        로그 파일을 분석하여 구조화된 데이터 반환
        """
        results = []
        
        try:
            with open(file_path, 'r', encoding='utf-8') as file:
                for line_num, line in enumerate(file, 1):
                    line = line.strip()
                    if line:  # 빈 줄 무시
                        log_info = self.extract_log_info(line)
                        log_info['line_number'] = line_num
                        results.append(log_info)
        
        except FileNotFoundError:
            print(f"파일을 찾을 수 없습니다: {file_path}")
        except Exception as e:
            print(f"파일 읽기 중 오류 발생: {e}")
        
        return results
    
    def generate_summary(self, log_data: List[Dict[str, str]]) -> Dict[str, int]:
        """
        로그 레벨별 통계 생성
        """
        summary = {}
        for entry in log_data:
            level = entry['level']
            summary[level] = summary.get(level, 0) + 1
        
        return summary

# 사용 예제
if __name__ == "__main__":
    analyzer = LogAnalyzer()
    
    # 샘플 로그 데이터
    sample_logs = [
        "2023-10-15 10:30:25 INFO (Auth) User john_doe logged in successfully",
        "2023-10-15 10:31:45 ERROR (Database) Connection timeout: 30 seconds",
        "2023-10-15 10:32:15 WARNING (System) CPU usage high: 95%",
        "2023-10-15 10:33:00 DEBUG (Network) Packet received from 192.168.1.100",
        "This is a log line without level or brackets"
    ]
    
    # 분석 실행
    analyzed_data = []
    for log_line in sample_logs:
        result = analyzer.extract_log_info(log_line)
        analyzed_data.append(result)
    
    # 결과 출력
    print("로그 분석 결과:")
    print("-" * 80)
    for entry in analyzed_data:
        print(f"레벨: {entry['level']:8} | 메시지: {entry['message']}")
    
    # 통계 생성
    summary = analyzer.generate_summary(analyzed_data)
    print("\n레벨별 통계:")
    print("-" * 30)
    for level, count in summary.items():
        print(f"{level}: {count}개")
```

## 4. 고급 기능: 타임스탬프 및 추가 정보 추출

```python
def enhanced_log_analysis(log_line: str) -> Dict[str, str]:
    """
    향상된 로그 분석 함수 - 타임스탬프, 모듈명 등 추가 정보 추출
    """
    # 타임스탬프 패턴 (YYYY-MM-DD HH:MM:SS 형식 가정)
    timestamp_pattern = r'(\d{4}-\d{2}-\d{2} \d{2}:\d{2}:\d{2})'
    level_pattern = r'\b(DEBUG|INFO|WARN|WARNING|ERROR|CRITICAL|FATAL)\b'
    module_pattern = r'\(([^)]+)\)'  # 괄호 안의 모듈명 추출
    
    result = {
        'timestamp': None,
        'level': 'UNKNOWN',
        'module': None,
        'message': log_line,
        'original': log_line
    }
    
    # 타임스탬프 추출
    timestamp_match = re.search(timestamp_pattern, log_line)
    if timestamp_match:
        result['timestamp'] = timestamp_match.group(1)
    
    # 레벨 추출
    level_match = re.search(level_pattern, log_line, re.IGNORECASE)
    if level_match:
        result['level'] = level_match.group(1).upper()
    
    # 모듈명 추출 (괄호 안의 내용)
    module_match = re.search(module_pattern, log_line)
    if module_match:
        result['module'] = module_match.group(1)
    
    return result

# 고급 분석 사용 예제
advanced_logs = [
    "2023-10-15 10:30:25 INFO (Authentication) User login successful",
    "2023-10-15 10:31:45 ERROR (Database.MySQL) Connection failed",
    "2023-10-15 10:32:15 WARN (System.Monitoring) High memory usage"
]

print("고급 로그 분석 결과:")
print("-" * 60)
for log in advanced_logs:
    analysis = enhanced_log_analysis(log)
    print(f"시간: {analysis['timestamp']}")
    print(f"레벨: {analysis['level']}")
    print(f"모듈: {analysis['module']}")
    print(f"메시지: {analysis['message']}")
    print("-" * 40)
```

## 5. 실제 파일 분석 예제

```python
def analyze_real_log_file(file_path: str):
    """
    실제 로그 파일 분석 예제
    """
    analyzer = LogAnalyzer()
    
    # 파일 분석
    log_data = analyzer.analyze_log_file(file_path)
    
    if log_data:
        # 결과를 pandas DataFrame으로 변환
        df = pd.DataFrame(log_data)
        
        # 기본 통계 출력
        print(f"총 로그 라인 수: {len(df)}")
        print("\n레벨별 분포:")
        print(df['level'].value_counts())
        
        # 에러 로그만 필터링
        error_logs = df[df['level'] == 'ERROR']
        if not error_logs.empty:
            print("\n에러 로그 목록:")
            for _, row in error_logs.iterrows():
                print(f"라인 {row['line_number']}: {row['message']}")
    
    return log_data
```

이 코드들은 다음과 같은 기능을 제공합니다:

1. **로그 레벨 추출**: 정규식을 사용하여 DEBUG, INFO, WARN, ERROR 등의 로그 레벨을 추출
2. **메시지 구분**: 첫 번째 괄호 이후의 문자열을 메시지로 자동 구분
3. **구조화된 데이터**: 분석 결과를 딕셔너리 형태로 반환
4. **통계 기능**: 레벨별 로그 수 집계
5. **확장성**: 타임스탬프, 모듈명 등 추가 정보 추출 가능

실제 사용 시에는 로그 형식에 맞게 정규식 패턴을 조정해야 합니다.
